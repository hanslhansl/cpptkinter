344 void PythonCmdImpl(Tcl_Interp* interp, int objc, ::Tcl_Obj* const objv[])
345 {
346 if (objc < 1)
347 throw utility::construct_exception<TclError>("this should never get triggered. objc should never be < 1 but is " + std::to_string(objc));
348
349 auto args = std::ranges::subrange(objv + 1, objv + objc) | std::views::transform([](const auto& a) { return Tcl_Obj(a); }) | std::ranges::to<std::vector>();
350
351 if constexpr (detail::PythonCmd_ClientDataArgsConcept2<Args...>)
352 ;//pass
353 else
354 {
355 if (sizeof...(Args) != args.size())
356 {
357 std::string error_string = std::format("{}\ngot {} tcl arguments:", this->get_error_string_header(objv[0]), args.size());
358 for (auto&& arg : args)
359 error_string += "\n\t- " + detail::Tcl_Obj_to_string(this->self, arg);
360 error_string += std::format("\nexpected {} c++ arguments:", sizeof...(Args));
361 error_string += ("" + ... + ("\n\t- " + std::string(reflect::type_name<Args>())));
362 throw utility::construct_exception<TclError>(error_string);
363 }
364 }
365
366 ENTER_PYTHON;
367
368 auto inner_caller = [&]<typename T, size_t I>() {
369 auto opt_result = detail::FromObjImpl(this->self, args[I], std::type_identity<std::remove_cvref_t<T>>{});
370 if (opt_result.has_value())
371 return std::move(*opt_result);
372
373 std::string error_string = std::format("{}\ntcl argument at position {} was {}\nexpected c++ argument type {}",
374 this->get_error_string_header(objv[0]),
375 I + 1,
376 detail::Tcl_Obj_to_string(this->self, args[I]),
377 reflect::type_name<T>());
378 throw utility::construct_exception<TclError>(error_string);
379 };
380
381 // necessary bc c++ doesn't define function argument evaluation order :(
382 auto caller = [&]<size_t...I>(std::index_sequence<I...>) {
383 if constexpr (detail::PythonCmd_ClientDataArgsConcept2<Args...>) // pass std::vector<Tcl_Obj>
384 this->func(std::move(args));
385 else // pass c++ args
386 return this->func(inner_caller.template operator() < Args, I > ()...);
387 };
388
389 if constexpr (std::same_as<R, void>)
390 {
391 caller(std::make_index_sequence<sizeof...(Args)>{});
392 Tcl_ResetResult(interp);
393 }
394 else
395 {
396 auto obj_res = AsObj(caller(std::make_index_sequence<sizeof...(Args)>{}));
397 Tcl_SetObjResult(interp, obj_res);
398 }
