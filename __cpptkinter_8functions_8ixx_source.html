272 std::string Tcl_Obj_to_string(TkappObjectImpl* self, ::Tcl_Obj* value)
273 {
274 auto type_string = Tcl_obj_type_string(value);
275
276 if (!value)
277 throw utility::construct_exception<TclError>(std::format("Tcl_Obj* is nullptr"));
278 else if (!value->typePtr)
279 return std::format("(unknown type)'{}'", Tcl_GetString(value));
280 else if (!value->typePtr->name)
281 throw utility::construct_exception<TclError>(std::format("Tcl_Obj->typePtr->name is nullptr (Tcl_GetString: {})", Tcl_GetString(value)));
282 else
283 {
284 std::string result{};
285
286 if (value->typePtr == self->DictType)
287 {
288 Tcl_Interp* interp = self->interp;
289 Tcl_DictSearch search{};
290 ::Tcl_Obj* keyPtr, * valuePtr;
291 int done{};
292 if (Tcl_DictObjFirst(interp, value, &search, &keyPtr, &valuePtr, &done) != TCL_OK)
293 throw Tkinter_Error(self);
294
295 result += type_string;
296 result += "{ ";
297
298 while (!done) {
299 result += Tcl_Obj_to_string(self, keyPtr) + " : " + Tcl_Obj_to_string(self, valuePtr) + ", ";
300
301 Tcl_DictObjNext(&search, &keyPtr, &valuePtr, &done);
302 }
303
304 Tcl_DictObjDone(&search);
305
306 result += "}";
307 }
308 else if (value->typePtr == self->ListType)
309 {
310 Tcl_Interp* interp = self->interp;
311 Tcl_Size size{};
312 if (Tcl_ListObjLength(interp, value, &size) == TCL_ERROR)
313 throw Tkinter_Error(self);
314
315 result += type_string;
316 result += "[ ";
317
318 for (Tcl_Size i = 0; i < size; i++)
319 {
320 ::Tcl_Obj* tcl_elem{};
321 if (Tcl_ListObjIndex(interp, value, i, &tcl_elem) == TCL_ERROR)
322 throw Tkinter_Error(self);
323 result += Tcl_Obj_to_string(self, tcl_elem);
324 if (i != size - 1)
325 result += ", ";
326 }
327
328 result += "]";
329 }
330 else
331 {
332 result += std::format("({})'{}'", type_string, Tcl_GetString(value));
333 }
334 return result;
335 }
