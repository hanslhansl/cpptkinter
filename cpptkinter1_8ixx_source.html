531{
532 auto index = name.find('.');
533 std::shared_ptr<impl> w;
534
535 if (index == 0)
536 {
537 w = this->_root().pimpl;
538 name.remove_prefix(1);
539 index = name.find('.');
540 }
541 else
542 w = this->pimpl;
543

546 if (current_name.empty())
